package main

import (
	"fmt"
	"os"

	"github.com/mxschmitt/playwright-go"
)

// 与 f_monitor 相同的浏览器解析策略：BROWSER_PATH 环境变量 > 固定路径 > channel。
var browserPriority = []string{
	`D:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe`,
	`C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe`,
}

var browserChannelPriority = []string{"msedge", "chrome"}

func resolveBrowser() (string, string) {
	if p := os.Getenv("BROWSER_PATH"); p != "" {
		if _, err := os.Stat(p); err == nil {
			return p, ""
		}
	}
	for _, p := range browserPriority {
		if _, err := os.Stat(p); err == nil {
			return p, ""
		}
	}
	return "", browserChannelPriority[0]
}

func main() {
	if err := run(); err != nil {
		fmt.Fprintln(os.Stderr, "FATAL:", err)
		os.Exit(1)
	}
}

func run() error {
	pw, err := playwright.Run()
	if err != nil {
		return fmt.Errorf("无法启动 Playwright driver: %w", err)
	}
	defer pw.Stop()

	execPath, channel := resolveBrowser()
	opts := playwright.BrowserTypeLaunchOptions{
		Headless:          playwright.Bool(false),
		IgnoreDefaultArgs: []string{"--enable-automation"},
		Args: []string{
			"--disable-blink-features=AutomationControlled",
			"--no-first-run",
			"--no-default-browser-check",
			"--lang=zh-CN",
		},
	}
	if execPath != "" {
		fmt.Println("LOG: 使用浏览器:", execPath)
		opts.ExecutablePath = &execPath
	} else {
		fmt.Println("LOG: 使用 channel:", channel)
		opts.Channel = &channel
	}
	browser, err := pw.Chromium.Launch(opts)
	if err != nil {
		return fmt.Errorf("启动浏览器失败: %w", err)
	}
	defer browser.Close()

	context, err := browser.NewContext(playwright.BrowserNewContextOptions{
		Locale:     playwright.String("zh-CN"),
		TimezoneId: playwright.String("Asia/Shanghai"),
		ExtraHttpHeaders: map[string]string{
			"Accept-Language": "zh-CN,zh;q=0.9,en;q=0.8",
		},
	})
	if err != nil {
		return fmt.Errorf("创建 context 失败: %w", err)
	}
	page, err := context.NewPage()
	if err != nil {
		return err
	}

	// 记录所有请求跳转链
	page.On("request", func(r playwright.Request) {
		fmt.Printf("REQ  %-4s %s\n", r.Method(), r.URL())
	})
	page.On("response", func(r playwright.Response) {
		loc := r.Headers()["location"]
		fmt.Printf("RESP %-4d %s loc=%s\n", r.Status(), r.URL(), loc)
	})

	fmt.Println("STEP: 打开 wfw 首页")
	if _, err := page.Goto("https://wfw.gdou.edu.cn/", playwright.PageGotoOptions{
		WaitUntil: playwright.WaitUntilStateDomcontentloaded,
	}); err != nil {
		return fmt.Errorf("打开 wfw 失败: %w", err)
	}
	fmt.Println("NOW URL:", page.URL())
	dumpCookies(context, "打开 wfw 后")

	fmt.Println("DONE — 浏览器保持打开，手动登录后按回车继续")
	fmt.Scanln()
	dumpCookies(context, "手动登录后")
	state, _ := context.StorageState()
	fmt.Printf("STATE: %s\n", state)
	return nil
}

func dumpCookies(context playwright.BrowserContext, tag string) {
	cookies, err := context.Cookies()
	if err != nil {
		fmt.Println("cookies error:", err)
		return
	}
	fmt.Printf("---- cookies @%s (%d) ----\n", tag, len(cookies))
	for _, c := range cookies {
		v := c.Value
		if len(v) > 12 {
			v = v[:6] + "..." + v[len(v)-4:]
		}
		fmt.Printf("  %-16s domain=%-28s path=%-6s value=%s\n", c.Name, c.Domain, c.Path, v)
	}
}
